\section{Scattering by a Central Potential}

% segments/page01_seg01.png
\subsection{Kinematics and Definitions}

\begin{align*}
\text{Incident energy: } & E = \frac{m}{2} v_0^2 \\
\text{Incident velocity: } & v_0 = \sqrt{2E/m} \\
\text{Angular momentum: } & \ell = s\, m v_0 = s\sqrt{2mE}, \qquad s = \frac{\ell}{\sqrt{2mE}}
\end{align*}

Here $s$ is the \emph{impact parameter}: particles are incident with impact parameter between
$s$ and $s+ds$, and travel along the incoming direction until they reach the apsidal point
(distance of closest approach) and emerge scattered through angle $\theta$ relative to the
incident direction. By azimuthal symmetry the scattering does not depend on the azimuthal angle
$\varphi$; only $\theta$ (and $E$) matter, so the scattered intensity distribution is written
$\sigma(\theta,E)$.

\begin{figure}[h]
\centering
\includegraphics[width=0.85\linewidth]{../segments/page01_seg01.png}
\caption{Incident-beam geometry: impact parameter $s$, apsidal point, and scattering angle
$\theta$. Volume element $d^3r = r^2\,dr\,d\Omega$ with $d\Omega = \sin\theta\,d\theta\,d\varphi$.}
\end{figure}

The solid angle into which particles are scattered is $d\Omega = 2\pi \sin\theta\, d\theta$
(azimuthal integration already performed, using the symmetry above).

% segments/page01_seg02.png
\subsection{Differential and Total Cross Section (General Definition)}

The incident ring of particles with impact parameter between $s$ and $s+ds$ has area
$2\pi s\,ds$. Define the incident intensity
\[
I = \frac{\#\ \text{particles incident per unit time}}{\text{incident beam cross-section area}},
\]
and the differential cross section $\sigma(\Omega)$ via
\[
\sigma(\Omega)\, d\Omega = \frac{\#\ \text{particles scattered into } d\Omega \text{ per unit time}}{\text{incident intensity}}.
\]

Particle-number conservation between the incident ring and the emerging angular range gives
\[
I (2\pi s)\,|ds| \;=\; \#\ \text{particles between } s,\, s+ds \;=\; \#\ \text{particles emerging between } \theta,\,\theta+d\theta \;=\; I\,\sigma(\theta)\, 2\pi \sin\theta\, |d\theta|.
\]

Therefore
\begin{equation}
\boxed{\sigma(\theta) = \frac{s}{\sin\theta}\left|\frac{ds}{d\theta}\right|}
\end{equation}
and the total cross section is defined as
\begin{equation}
\boxed{\sigma_T = \int d\Omega\, \sigma(\Omega).}
\end{equation}

% segments/page01_seg03.png
\subsection{Relating $\theta$ and $\psi$: Repulsive and Attractive Cases}

Let $\psi$ be the angle swept from the incident asymptote to the apsidal point. Two geometric
cases arise depending on whether the interaction is repulsive or attractive:

\textbf{Repulsive case:}
\[
\pi = \theta + 2\psi \quad\Longrightarrow\quad \theta = \pi - 2\psi, \qquad \psi = \frac{\pi-\theta}{2}.
\]

\textbf{Attractive case:}
\[
\pi = 2\psi - \theta \quad\Longrightarrow\quad \theta = 2\psi - \pi, \qquad \psi = \frac{\pi+\theta}{2}.
\]

In all cases:
\begin{equation}
\boxed{\theta = |\pi - 2\psi|}
\end{equation}

\centering
\includegraphics[width=0.85\linewidth]{../segments/page01_seg03.png}
\caption{Repulsive-case (left) and attractive-case (right) scattering geometry, relating the
apsidal-point sweep angle $\psi$ to the scattering angle $\theta$.}
\end{figure}

\subsection{Computing $S(\theta,E)$: the Universal Method}

Starting from energy conservation with the centrifugal term,
\[
E = \frac{m}{2}\dot r^2 + V(r) + \frac{\ell^2}{2mr^2}
\quad\Longrightarrow\quad
\dot r = \left[\frac{2}{m}\left(E - V(r) - \frac{\ell^2}{2mr^2}\right)\right]^{1/2},
\qquad \dot r = \frac{dr}{dt} \Rightarrow dt = \frac{dr}{\dot r}.
\]

From $\ell = m r^2 \dot\theta$,
\[
d\theta = \dot\theta\, dt = \frac{\ell}{mr^2}\,dt = \frac{\ell}{mr^2}\cdot\frac{dr}{\dot r}
= \frac{\ell}{m}\sqrt{\frac{m}{2E}}\; \frac{r^{-2}\,dr}{\left[1 - \dfrac{V(r)}{E} - \dfrac{\ell^2}{2mEr^2}\right]^{1/2}},
\]
where $\ell/\sqrt{2mE} = s$ and $\ell^2/(2mE) = s^2$ (both from the definition of $s$ above), so
\begin{equation}
d\theta = \frac{s\,r^{-2}\,dr}{\left[1 - \dfrac{V(r)}{E} - \dfrac{s^2}{r^2}\right]^{1/2}}.
\end{equation}
% REVIEW: the notes' intermediate line writes the bracket as "1 - V(r) - s^2/r^2" (dropping the
% /E on V(r)); the /E is restored two lines later in the boxed theta(s) formula below. Transcribed
% here with /E included throughout for internal consistency, per the corrected/boxed version.

Integrating from the apsidal point ($r=r_m$, $\theta=\theta_m$) out to $r\to\infty$ ($\theta\to\pi$):
\[
\psi = \int_0^\psi d\psi = \int_{\theta_m}^{\pi} d\theta = \int_{r_m}^{\infty} \frac{s\,r^{-2}\,dr}{\left[1-\dfrac{V(r)}{E}-\dfrac{s^2}{r^2}\right]^{1/2}}.
\]

Substituting $u = 1/r$, $du = -r^{-2}\,dr$, $u_m = 1/r_m$, and using $\theta = |\pi-2\psi|$:
\begin{equation}
\boxed{\theta(s) = \left|\pi - 2\int_0^{u_m} \frac{s\,du}{\left[1 - \dfrac{V(1/u)}{E} - s^2u^2\right]^{1/2}}\right|}
\end{equation}

From here one computes $\left|\dfrac{ds}{d\theta}\right| = 1\Big/\left|\dfrac{d\theta}{ds}\right|$
and uses it to get $\sigma(\theta) = \dfrac{s}{\sin\theta}\left|\dfrac{ds}{d\theta}\right|$.

\begin{remark}
Up to here, all results are valid for \emph{any} central potential $V(r)$.
\end{remark}

\section{Coulomb (Rutherford) Scattering}

% segments/page02_seg01.png
\subsection{Setup: Orbit Equation and Eccentricity}

For the Coulomb force between charges $Ze$ and $Z'e$,
\[
f = \frac{ZZ'e^2}{r^2}, \qquad V(r) = -\frac{k}{r}, \qquad k = -ZZ'e^2.
\]

\begin{center}
\begin{tabular}{ll}
$E>0$ & hyperbolic \\
$E=0$ & parabolic \\
$E<0$ & elliptic
\end{tabular}
\quad (only for $V = -k/r$)
\end{center}

For scattering we take $E>0$ (hyperbolic orbit). The orbit equation is
\[
\frac{1}{r} = C_1\left[1 + \epsilon\cos(\tilde\theta - \tilde\theta')\right], \qquad
C_1 = \frac{mk}{\ell^2} = -\frac{m}{\ell^2}ZZ'e^2,
\]
with eccentricity
\[
\epsilon = \left[1 + \frac{2E\ell^2}{mk^2}\right]^{1/2}
= \left[1 + \frac{2E\ell^2}{m(ZZ'e^2)^2}\right]^{1/2}
= \left[1 + \left(\frac{2sE}{ZZ'e^2}\right)^2\right]^{1/2} > 1,
\]
using $\ell = s m v_0 = s\sqrt{2mE}$, so $\ell^2 = 2mEs^2$ and $2E\ell^2/(mk^2) = (2sE/k)^2$.

Two situations arise:
\begin{align*}
\text{i) Repulsive case: } & ZZ' > 0 \Rightarrow k = -ZZ'e^2 < 0 \\
\text{ii) Attractive case: } & ZZ' < 0 \Rightarrow k = -ZZ'e^2 > 0 \quad (\text{also gravity, } k = Gm_1m_2)
\end{align*}

% segments/page02_seg02.png
\subsection{Repulsive Case}

Here $k<0$, so $C_1 = mk/\ell^2 < 0$. Physical orbits require $1/r>0$, i.e.
$C_1[1+\epsilon\cos(\tilde\theta-\tilde\theta')] > 0 \Leftrightarrow \cos(\tilde\theta-\tilde\theta') < -1/\epsilon$:
we need the cosine to be negative and larger in magnitude than $1/\epsilon$.

$\tilde\theta = \tilde\theta'$ is not a physical point, since $\cos(\tilde\theta'-\tilde\theta')=\cos 0=1$
would give $1/r = C_1(1+\epsilon) < 0$ (unphysical). So $\tilde\theta=\tilde\theta'$ lies in the
middle of the inaccessible region of the hyperbola, and $\tilde\theta = \tilde\theta' \pm \pi$ is
the periapsis; choose $\tilde\theta' = \pi$:
\[
\cos(\tilde\theta - \tilde\theta') = \cos(\tilde\theta-\pi) = \cos\tilde\theta\cos\pi + \sin\tilde\theta\sin\pi = -\cos\tilde\theta,
\]
so
\[
\frac{1}{r} = \left(-\frac{mk}{\ell^2}\right)\left[-1-\epsilon\cos(\tilde\theta-\tilde\theta')\right]
= \frac{mZZ'e^2}{\ell^2}\left[\epsilon\cos\tilde\theta - 1\right].
\]
Maximum $1/r$ (minimum $r$) occurs at $\tilde\theta=0$:
$\dfrac{1}{r_m} = \dfrac{mZZ'e^2}{\ell^2}[\epsilon-1]$.

The apsidal angle is $\tilde\theta = 0$; we need $\psi = \tilde\theta(r\to\infty) = \tilde\theta(1/r\to0)$:
\[
0 = \epsilon\cos\psi - 1 \Leftrightarrow \frac{1}{\epsilon} = \cos\psi = \cos\left(\frac{\pi-\theta}{2}\right)
= \cos\frac{\pi}{2}\cos\frac{\theta}{2} + \sin\frac{\pi}{2}\sin\frac{\theta}{2} = \sin\frac{\theta}{2}
\]
(using the repulsive-case relation $\theta = \pi-2\psi \Rightarrow \psi = (\pi-\theta)/2$).

% segments/page02_seg03.png
\subsection{Attractive Case}

Here $k>0$, so $C_1 = km/\ell^2 > 0$. Take $\tilde\theta'=0$, so at $\tilde\theta=\tilde\theta'=0$:
$1/r = C_1[1+\epsilon\cos(0)] = C_1[1+\epsilon]$. The apsidal point is also at $\tilde\theta=0$;
we again need $\psi = \tilde\theta(r\to\infty) = \tilde\theta(1/r\to0)$.

Maximum $1/r$ (minimum $r$) is at $\tilde\theta=0$: $\dfrac{1}{r_m} = C_1[1+\epsilon]$.

\[
0 = 1+\epsilon\cos\psi \Leftrightarrow -\frac{1}{\epsilon} = \cos\psi = \cos\left(\frac{\pi+\theta}{2}\right)
= \cos\frac{\pi}{2}\cos\frac{\theta}{2} - \sin\frac{\pi}{2}\sin\frac{\theta}{2} = -\sin\frac{\theta}{2}
\]
(using the attractive-case relation $\theta = 2\psi-\pi \Rightarrow \psi = (\pi+\theta)/2$).

\section{Combining Both Cases: $S(\theta,E)$ and the Rutherford Cross Section}

% segments/page03_seg01.png
\subsection{The Impact Parameter $S(\theta,E)$}

Both cases give $\cos\psi = \pm\sin(\theta/2)$, i.e.\ $\cot^2(\theta/2) = 1/\sin^2(\theta/2) - 1$.
Combining with $\cos\psi = 1/\epsilon$ (repulsive) or $-1/\epsilon = \cos\psi$ (attractive), in both
cases $\cos^2\psi = 1/\epsilon^2$, and $\sin^2(\theta/2) = 1/\epsilon^2$ gives
\[
\cot^2\frac{\theta}{2} = \frac{1}{\tan^2\frac\theta2} = \frac{\cos^2\theta/2}{\sin^2\theta/2}
= \frac{1-\sin^2\theta/2}{\sin^2\theta/2} = \frac{1}{\sin^2\theta/2}-1 = \epsilon^2-1 = \left(\frac{2sE}{ZZ'e^2}\right)^2,
\]
so
\begin{equation}
\boxed{s = S(\theta,E) = \left|\frac{ZZ'e^2}{2E}\right|\cot\frac{\theta}{2}}
\end{equation}

% segments/page03_seg02.png
\subsection{The Rutherford Differential Cross Section}

\[
\frac{ds}{d\theta} = \left|\frac{ZZ'e^2}{2E}\right|\frac{d}{d\theta}\left(\cot\frac\theta2\right)
= \left|\frac{ZZ'e^2}{2E}\right|\left(-\frac{1}{2}\frac{1}{\sin^2\theta/2}\right)
= -\left|\frac{ZZ'e^2}{4E}\right|\frac{1}{\sin^2\theta/2}.
\]
Then
\[
\sigma(\theta) = \frac{s}{\sin\theta}\left|\frac{ds}{d\theta}\right|
= \frac12\left(\frac{ZZ'e^2}{2E}\right)^2 \frac{1}{2\sin\frac\theta2\cos\frac\theta2}\cdot\frac{\cos\frac\theta2}{\sin\frac\theta2}\cdot\frac{1}{\sin^2\frac\theta2},
\]
which simplifies to the \textbf{Rutherford scattering cross section (for Coulomb forces)}:
\begin{equation}
\boxed{\sigma(\theta) = \frac14\left(\frac{ZZ'e^2}{2E}\right)^2 \csc^4\left(\frac\theta2\right)}
\end{equation}

% segments/page03_seg03.png
\subsection{Remarks}
\begin{enumerate}
\item The cross section is the same for the repulsive and attractive cases, since it is
proportional to $(ZZ')^2$.
\item It diverges like $E^{-2}$ at low energies.
\item It diverges like $\theta^{-4}$ (very strong!) for small angles $\theta$.
\item This classical result is exactly the same as the result obtained in Quantum Mechanics.
\end{enumerate}

% segments/page03_seg04.png
\subsection{Total Scattering Cross Section}

\[
\sigma_T = \int \sigma(\Omega)\, d\Omega.
\]

\begin{remark}
Notation: Goldstein writes $\sigma(\Omega) \leftrightarrow d\sigma/d\Omega$ (differential
scattering cross section) and $\sigma_T \leftrightarrow \sigma$ (total scattering cross section);
other texts may use different symbols for the same quantities.
\end{remark}

% segments/page03_seg05.png
In this case $\sigma(\theta) = \mathrm{const}\times \dfrac{1}{\sin^4(\theta/2)}$, and
$d\Omega = 2\pi\sin\theta\, d\theta = 4\pi\sin\frac\theta2\cos\frac\theta2\, d\theta$, so
\[
\sigma_T = 4\pi\cdot\frac14\left(\frac{ZZ'e^2}{2E}\right)^2\int_0^\pi \frac{\sin\frac\theta2\cos\frac\theta2}{\sin^4\frac\theta2}\, d\theta
\sim \int_0^\delta \frac{d\theta}{(\theta/2)^3} = 8\int_0^\delta \frac{d\theta}{\theta^3}
= 8\,\frac{\theta^{-2}}{-2}\Big|_0^\delta = +\infty \quad (\text{strongly divergent}).
\]

\begin{remark}
Classically, $\sigma_T = +\infty$ for \emph{any} long-range potential. In Quantum Mechanics,
$\sigma_T=+\infty$ only for those long-range potentials $V(r)$ that do not decay fast enough as
$r\to\infty$.
\end{remark}

% segments/page03_seg06.png
\subsection{Caveat: Screened Coulomb (Yukawa) Potential}

In a dense conducting material, the bare Coulomb potential $V=-k/r$ is screened:
\[
V(r) = -\frac{k}{r} \;\longrightarrow\; V^{\mathrm{screen}}(r) = -\frac{k}{r}\,e^{-r/\lambda}
\qquad \text{(screened Coulomb potential, ``Yukawa'')}.
\]
(In non-conducting materials $V^{\mathrm{screen}}(r)$ may take a different form, but will still
vanish much faster than $1/r$.)

\section{Chapter 8: Hamiltonian and Hamilton's Equations of Motion}

% segments/page04_seg01.png
\subsection{Lagrangian and the Euler--Lagrange Equations}

\[
L = T - V = L(q_1,\dots,q_n,\dot q_1,\dots,\dot q_n,t),
\]
\[
0 = \left.\frac{d}{dt}\left(\frac{\partial L}{\partial \dot q_i}\right)\right|_{q_j,\dot q_j} - \left.\frac{\partial L}{\partial q_i}\right|_{q_j,\dot q_j},
\qquad
p_i = \left.\frac{\partial L}{\partial \dot q_i}\right|_{q_j,\dot q_j}.
\]

\begin{remark}
\textbf{Midterm remark (marked invalid in the original notes, crossed out):} in polar
coordinates one might be tempted to write $\partial L/\partial\dot\theta = p_\theta = \ell_z$ and
claim $L = L(r,\dot r, p_\theta)$, treating $p_\theta$ as an independent variable of the
Lagrangian and applying the Lagrange equations to it directly. This is invalid — $L$ is a
function of coordinates and \emph{velocities} $(q_i,\dot q_i)$, not momenta; conflating $p_\theta$
with a Lagrangian argument is exactly the error that the Legendre transformation to the
Hamiltonian (below) is needed to correctly resolve. The professor crossed this out in the
original notes as an invalid claim, rather than erasing it, precisely to flag it as a common
mistake.
\end{remark}

% segments/page04_seg02.png
\subsection{Legendre Transformation: a Thermodynamic Analogy}

The Legendre transformation exchanges the role of a variable with its conjugate "slope"
variable: for $q_1,\dots,q_n,\dot q_1,\dots,\dot q_n \to q_1,\dots,q_n,p_1,\dots,p_n$, recall for
a function of two variables $f(x,y)$: $df = \dfrac{\partial f}{\partial x}dx + \dfrac{\partial f}{\partial y}dy$.

In Thermodynamics (fluid at constant particle number $N$), the First Law reads
\[
dU = \delta Q + \delta W = T\,dS - p\,dV, \qquad U = U(S,V) \ \text{(internal energy)}.
\]
If instead we want $(T,V)$ as the natural variables, define the Helmholtz free energy
\[
F = A = U - TS,
\]
so that
\[
dA = dU - d(TS) = (T\,dS - p\,dV) - T\,dS - S\,dT = -S\,dT - p\,dV \quad\Longrightarrow\quad A = A(T,V).
\]

% segments/page04_seg03.png
\subsection{General Legendre Transformation}

For a general $f(x,y)$ with $df = u\,dx + v\,dy$, $u = \partial f/\partial x$, $v=\partial f/\partial y$,
to swap $(x,y)\to(u,y)$ define
\[
g = f - ux \quad\Longrightarrow\quad dg = df - d(ux) = (u\,dx+v\,dy) - u\,dx - x\,du = -x\,du + v\,dy,
\]
so $g = g(u,y)$.

% segments/page04_seg04.png -- page04_seg05.png
\subsection{The Hamiltonian and Canonical Equations of Motion}

Apply this to $L(q_1,\dots,q_n,\dot q_1,\dots,\dot q_n)$ with $p_i = \left.\dfrac{\partial L}{\partial \dot q_i}\right|_{q_j,\dot q_j}$
playing the role of the conjugate variable to $\dot q_i$. Define
\[
H = -\left(L - \sum_i p_i \dot q_i\right).
\]
Then
\[
dH = -dL + \sum_i d(p_i \dot q_i)
= -\sum_i\left(\frac{\partial L}{\partial q_i}dq_i + \frac{\partial L}{\partial \dot q_i}d\dot q_i\right) - \frac{\partial L}{\partial t}dt+ \sum_i\left(p_i\, d\dot q_i + \dot q_i\, dp_i\right),
\]
and using $p_i = \partial L/\partial \dot q_i$ the $d\dot q_i$ terms cancel, leaving
\[
dH = -\frac{\partial L}{\partial t}dt + \sum_i\left(\dot q_i\, dp_i - \frac{\partial L}{\partial q_i}dq_i\right)
\quad\Longrightarrow\quad H = H(p_1,\dots,p_n,q_1,\dots,q_n,t).
\]

% segments/page04_seg06.png
Comparing coefficients of $dp_i$, $dq_i$, $dt$ and using the Euler--Lagrange equations
($\partial L/\partial q_i = \dot p_i$) gives the \textbf{canonical equations of motion}:
\begin{equation}
\boxed{\frac{\partial H}{\partial p_i} = \dot q_i, \qquad
\frac{\partial H}{\partial q_i} = -\frac{\partial L}{\partial q_i} = -\dot p_i, \qquad
\frac{\partial H}{\partial t} = -\frac{\partial L}{\partial t}.}
\end{equation}
